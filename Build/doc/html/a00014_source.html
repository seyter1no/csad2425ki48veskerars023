296int minimax(char currentPlayer, char aiPlayer, int depth) {
297 if (checkWin(aiPlayer)) {
298 return 10 - depth; // AI wins
299 } else if (checkWin(opponent(aiPlayer))) {
300 return depth - 10; // Opponent wins
301 } else if (isBoardFull()) {
302 return 0; // Draw
303 }
304 int bestScore = (currentPlayer == aiPlayer) ? -1000 : 1000;
305
306 for (int i = 0; i < BOARD_SIZE; i++) {
307 for (int j = 0; j < BOARD_SIZE; j++) {
308 if (board[i][j] != PLAYER_X && board[i][j] != PLAYER_O) {
309 board[i][j] = currentPlayer; // Make the move
310 int score = minimax(opponent(currentPlayer), aiPlayer, depth + 1);
311 board[i][j] = '1' + (i * BOARD_SIZE + j); // Reset the move
312 if (currentPlayer == aiPlayer) {
313 if (score > bestScore) {
314 bestScore = score;
315 }
316 } else {
317 if (score < bestScore)
318 {
319 bestScore = score;
320 }
321 }
322 }
323 }
324 }
325 return bestScore;
